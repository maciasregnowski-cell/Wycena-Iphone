  <script>
    const f = document.getElementById('quoteForm');
    const previews = document.getElementById('previews');
    const progress = document.getElementById('progress');
    const summaryBox = document.getElementById('summary');
    const showSummaryBtn = document.getElementById('showSummary');
    const successBox = document.getElementById('success');
    const maxFiles = 10, maxSize = 10 * 1024 * 1024;

    // Podgląd miniatur i usuwanie
    const fileInput = document.getElementById('photos');
    let files = [];
    fileInput.addEventListener('change', (e) => {
      const incoming = Array.from(e.target.files || []);
      const combined = [...files, ...incoming];
      const unique = combined.slice(0, maxFiles);
      const valid = [];
      previews.innerHTML = '';
      unique.forEach((file, idx) => {
        if (!file.type.startsWith('image/')) return;
        if (file.size > maxSize) return;
        valid.push(file);
        const url = URL.createObjectURL(file);
        const div = document.createElement('div');
        div.className = 'thumb';
        div.innerHTML = '<img alt="foto" src="'+url+'"><button type="button" class="rm">×</button>';
        previews.appendChild(div);
        div.querySelector('.rm').addEventListener('click', () => {
          valid.splice(idx, 1);
          files = valid;
          renderPreviews();
        });
      });
      files = valid;
      validatePhotos();
    });

    function renderPreviews(){
      previews.innerHTML = '';
      files.forEach((file, i) => {
        const url = URL.createObjectURL(file);
        const div = document.createElement('div');
        div.className = 'thumb';
        div.innerHTML = '<img alt="foto" src="'+url+'"><button type="button" class="rm">×</button>';
        previews.appendChild(div);
        div.querySelector('.rm').addEventListener('click', () => {
          files.splice(i,1);
          renderPreviews();
        });
      });
    }

    // Walidacja
    function setError(name, msg){
      const el = document.querySelector('.error[data-for="'+name+'"]');
      if (!el) return;
      if (msg){ el.textContent = msg; el.style.display = 'block'; }
      else { el.style.display = 'none'; }
    }
    function validateRequired(id, key){
      const el = document.getElementById(id);
      if (!el.value.trim()){ setError(key, el.tagName === 'SELECT' ? 'Wybierz opcję.' : 'To pole jest wymagane.'); return false; }
      setError(key, null); return true;
    }
    function validateContact(){
      const phone = document.getElementById('phone').value.trim();
      const insta = document.getElementById('instagram').value.trim();
      const ok = !!(phone || insta);
      setError('contact', ok ? null : 'Podaj numer telefonu lub Instagram.');
      return ok;
    }
    function validatePhotos(){
      const err = [];
      if (files.length > maxFiles) err.push('Maksymalnie ' + maxFiles + ' zdjęć.');
      if (files.some(f => f.size > maxSize)) err.push('Każde zdjęcie do 10 MB.');
      setError('photos', err.length ? err.join(' ') : null);
      return err.length === 0;
    }
    function formProgress(){
      const required = ['brand','capacity','condition','icloud'];
      const have = required.filter(id => document.getElementById(id).value.trim()).length + (validateContact()?1:0);
      const total = required.length + 1;
      const pct = Math.round(100 * have / total);
      progress.style.width = pct + '%';
    }
    f.addEventListener('input', formProgress);
    f.addEventListener('change', formProgress);
    window.addEventListener('load', formProgress);

    // Podsumowanie
    function buildSummary(){
      const data = collectData();
      const lines = [
        'Model: ' + data.brand,
        'Pojemność: ' + data.capacity,
        'Bateria: ' + (data.battery || '-'),
        'Stan: ' + data.condition,
        'iCloud/FRP: ' + data.icloud,
        'Akcesoria: ' + (data.box || '-'),
        'Usterki: ' + (data.issues || '-'),
        'Kontakt: ' + (data.phone ? 'tel: ' + data.phone : '') + (data.instagram ? (data.phone ? ', ' : '') + 'IG: ' + data.instagram : ''),
        'Zdjęcia: ' + files.length + ' szt.'
      ];
      return lines.join('\n');
    }
    showSummaryBtn.addEventListener('click', () => {
      summaryBox.textContent = buildSummary();
      summaryBox.style.display = 'block';
    });

    function collectData(){
      const data = Object.fromEntries(new FormData(f).entries());
      return data;
    }

    // Wysyłka — przykładowo do webhooka (np. Make/Zapier/wasz backend)
    async function sendData(){
      const data = collectData();
      // Dołączamy pliki jako FormData
      const fd = new FormData();
      Object.entries(data).forEach(([k,v]) => fd.append(k, v));
      files.forEach((file, i) => fd.append('photos', file, file.name || ('photo_'+(i+1)+'.jpg')));
      // PRZYKŁAD: podmień poniżej URL na swój endpoint (HTTPS)
      const ENDPOINT = '[example.com](https://example.com/api/wycena)'; // <- podmień
      const res = await fetch(ENDPOINT, { method:'POST', body: fd });
      if (!res.ok) throw new Error('Błąd wysyłki (' + res.status + ')');
      return await res.json().catch(() => ({}));
    }

    // Obsługa submit
    f.addEventListener('submit', async (e) => {
      e.preventDefault();
      const v1 = validateRequired('brand','brand');
      const v2 = validateRequired('capacity','capacity');
      const v3 = validateRequired('condition','condition');
      const v4 = validateRequired('icloud','icloud');
      const v5 = validateContact();
      const v6 = validatePhotos();
      if (!(v1 && v2 && v3 && v4 && v5 && v6)) return;

      // Blokuj przyciski i pokaż progres
      const btns = f.querySelectorAll('button');
      btns.forEach(b => b.disabled = true);
      progress.style.width = '100%';

      try {
        await sendData();
        successBox.style.display = 'block';
        f.reset();
        files = [];
        previews.innerHTML = '';
        summaryBox.style.display = 'none';
        formProgress();
      } catch (err) {
        alert('Nie udało się wysłać formularza. Sprawdź połączenie lub spróbuj ponownie.');
        console.error(err);
      } finally {
        btns.forEach(b => b.disabled = false);
        setTimeout(() => { progress.style.width = '0%'; }, 600);
      }
    });
  </script>
